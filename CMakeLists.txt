cmake_minimum_required(VERSION 3.25)
project(OBSBluray VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 is required")
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/scripts/msvc-launcher.py")
set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(SDK_ROOT "$ENV{LOCALAPPDATA}/StreamDeskSDK" CACHE PATH "Private build SDK directory")
set(GST_ROOT "${SDK_ROOT}/gstreamer" CACHE PATH "GStreamer MSVC SDK")
set(BLURAY_ROOT "${SDK_ROOT}/libbluray-1.4.0" CACHE PATH "libbluray source")
set(BLURAY_BUILD "${SDK_ROOT}/libbluray-build" CACHE PATH "libbluray build")
set(OBS_SDK_ROOT "${SDK_ROOT}/obs-sdk/obs-studio-32.2.2/libobs" CACHE PATH "Matching OBS headers")
set(OBS_IMPORT_LIBRARY "${SDK_ROOT}/obs-sdk/obs.lib" CACHE FILEPATH "OBS import library")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Concurrent Test)
add_library(bluray_core STATIC src/configuration.cpp src/common.cpp src/catalog.cpp src/pipeline.cpp src/subtitles.cpp src/transport.cpp src/bluray_audio.cpp src/media_preview.cpp)
target_include_directories(bluray_core PUBLIC src "${GST_ROOT}/include" "${GST_ROOT}/include/gstreamer-1.0" "${GST_ROOT}/lib/gstreamer-1.0/include" "${GST_ROOT}/include/glib-2.0" "${GST_ROOT}/lib/glib-2.0/include" "${BLURAY_ROOT}/src")
target_link_directories(bluray_core PUBLIC "${GST_ROOT}/lib" "${BLURAY_BUILD}/src")
target_compile_definitions(bluray_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN STREAMDESK_VERSION="${PROJECT_VERSION}")
target_compile_options(bluray_core PUBLIC /utf-8 /W4 /permissive- /Zc:__cplusplus)
target_link_libraries(bluray_core PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent gstreamer-1.0 gobject-2.0 glib-2.0 gstapp-1.0 gstvideo-1.0 gstaudio-1.0 gstd3d11-1.0 gstpbutils-1.0 bluray crypt32 ass d3d11 dxgi d3dcompiler)
add_executable(obs-bluray-worker obs-plugin/worker.cpp)
target_include_directories(obs-bluray-worker PRIVATE obs-plugin)
target_link_libraries(obs-bluray-worker PRIVATE bluray_core)
if(NOT EXISTS "${OBS_IMPORT_LIBRARY}")
  message(FATAL_ERROR "Run scripts/setup-obs-sdk.ps1 for your OBS installation")
endif()
add_library(obs-bluray MODULE obs-plugin/plugin.cpp)
set_target_properties(obs-bluray PROPERTIES AUTOMOC OFF PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(obs-bluray PRIVATE "${OBS_SDK_ROOT}" obs-plugin)
target_compile_definitions(obs-bluray PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(obs-bluray PRIVATE /utf-8 /W4)
target_link_libraries(obs-bluray PRIVATE "${OBS_IMPORT_LIBRARY}" ole32)
add_executable(obs-source-tests obs-plugin/host_test.cpp)
set_target_properties(obs-source-tests PROPERTIES AUTOMOC OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/obs-tests")
target_include_directories(obs-source-tests PRIVATE "${OBS_SDK_ROOT}" obs-plugin)
target_compile_definitions(obs-source-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(obs-source-tests PRIVATE /utf-8)
target_link_libraries(obs-source-tests PRIVATE "${OBS_IMPORT_LIBRARY}")
enable_testing()
add_executable(bluray-core-tests tests/core_test.cpp)
target_link_libraries(bluray-core-tests PRIVATE bluray_core Qt6::Test)
add_test(NAME core COMMAND bluray-core-tests)
